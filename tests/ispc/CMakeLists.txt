cmake_minimum_required(VERSION 3.24)
project(CyDSP_ISPC_Experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ISPC_EXECUTABLE "" CACHE FILEPATH "Explicit path to the ISPC compiler")
set(CYDSP_ARCHIVE "" CACHE FILEPATH "Release CyDSP static library matching this compiler ABI/CRT")
set(CYDSP_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../include" CACHE PATH "CyDSP public headers")
if(NOT EXISTS "${ISPC_EXECUTABLE}" OR NOT EXISTS "${CYDSP_ARCHIVE}")
 message(FATAL_ERROR "Pass ISPC_EXECUTABLE and CYDSP_ARCHIVE explicitly; no downloads or production integration occur")
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Fold.obj" "${CMAKE_CURRENT_BINARY_DIR}/Fold.h"
 COMMAND "${ISPC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Fold.ispc" --arch=x86-64 --target=avx2-i32x8 --opt=disable-fma -O2 -o "${CMAKE_CURRENT_BINARY_DIR}/Fold.obj" -h "${CMAKE_CURRENT_BINARY_DIR}/Fold.h"
 DEPENDS Fold.ispc VERBATIM)
add_executable(cydsp_ispc_comparison Comparison.cpp "${CMAKE_CURRENT_BINARY_DIR}/Fold.obj")
target_include_directories(cydsp_ispc_comparison PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CYDSP_INCLUDE}")
if(MSVC)
 target_compile_options(cydsp_ispc_comparison PRIVATE /fp:strict /O2)
else()
 target_compile_options(cydsp_ispc_comparison PRIVATE -O2 -ffp-contract=off)
endif()
target_link_libraries(cydsp_ispc_comparison PRIVATE "${CYDSP_ARCHIVE}")
